 * WHAT THIS DOES

/**
 * Resolves a ticker (e.g. "BHP.AX" or "BHP") to a logo image URL.
 * Uses the curated domain map, then fetches that company's own site favicon via
 * Google's public favicon service (the same one used elsewhere in this app — reliably
 * reachable, unlike Clearbit's logo API which has become unreachable/DNS-dead for some
 * networks since Clearbit's acquisition by HubSpot).
 */
function resolveLogo(ticker){
  if (!ticker) return null;
  var t = String(ticker).toUpperCase().replace(/\.AX$/, "").trim();
  var domain = TICKER_DOMAINS[t];
  if (domain){
    return "https://www.google.com/s2/favicons?sz=128&domain=" + encodeURIComponent(domain);
  }
  // Not in the curated map — no reliable domain to go on, so return nothing rather than
  // guess wildly (the client falls back to the manual Website field / initials avatar).
  return null;
}
